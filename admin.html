// Nexora Mobile — Admin Dashboard // Complete admin.js const app = document.getElementById("app"); const ADMIN_EMAIL = "sufiyanolawale36@gmail.com"; // Storage keys const USERS_KEY = "nexora_users"; const TRANSACTIONS_KEY = "nexora_transactions"; // ----------------------------- // STORAGE // ----------------------------- function getUsers() { try { return JSON.parse(localStorage.getItem(USERS_KEY)) || []; } catch { return []; } } function saveUsers(users) { localStorage.setItem(USERS_KEY, JSON.stringify(users)); } function getTransactions() { try { return JSON.parse(localStorage.getItem(TRANSACTIONS_KEY)) || []; } catch { return []; } } function saveTransactions(transactions) { localStorage.setItem( TRANSACTIONS_KEY, JSON.stringify(transactions) ); } // ----------------------------- // HELPERS // ----------------------------- function money(amount) { return "₦" + Number(amount || 0).toLocaleString("en-NG"); } function escapeHTML(value) { return String(value ?? "") .replaceAll("&", "&") .replaceAll("<", "<") .replaceAll(">", ">") .replaceAll('"', """) .replaceAll("'", "'"); } function createID(prefix = "NX") { return ( prefix + Date.now().toString(36) + Math.random().toString(36).substring(2, 7) ).toUpperCase(); } function formatDate(date) { return new Date(date).toLocaleString("en-NG", { dateStyle: "medium", timeStyle: "short" }); } // ----------------------------- // SAMPLE DATA // ----------------------------- function createDemoData() { let users = getUsers(); if (users.length === 0) { users = [ { id: createID("USR"), name: "Sufiyan Abdul afeez", email: "bcashapp119@gmail.com", wallet: 0, status: "active", joined: new Date().toISOString() }, { id: createID("USR"), name: "Olawale", email: "a66327810@gmail.com", wallet: 1500, status: "active", joined: new Date().toISOString() } ]; saveUsers(users); } if (!localStorage.getItem(TRANSACTIONS_KEY)) { saveTransactions([]); } } // ----------------------------- // DASHBOARD // ----------------------------- function renderDashboard() { const users = getUsers(); const transactions = getTransactions(); const totalWallet = users.reduce( (sum, user) => sum + Number(user.wallet || 0), 0 ); const totalTransactions = transactions.length; const completed = transactions.filter( t => t.status === "completed" ).length; const pending = transactions.filter( t => t.status === "pending" ).length; app.innerHTML = `

Nexora Admin Dashboard

Admin: ${escapeHTML(ADMIN_EMAIL)}

Dashboard Overview

Total Users: ${users.length}

Total Wallet Balance: ${money(totalWallet)}

Total Transactions: ${totalTransactions}

Completed: ${completed}

Pending: ${pending}

Quick Actions

Recent Transactions

${ transactions.length === 0 ? "

No transactions yet.

" : transactions .slice() .reverse() .slice(0, 5) .map(transactionHTML) .join("") }
`; } // ----------------------------- // USER MANAGEMENT // ----------------------------- function showUsers() { const users = getUsers(); app.innerHTML = `

Users

${users.map(userHTML).join("")}
`; } function filterUsers() { const search = document .getElementById("userSearch") .value .toLowerCase(); const users = getUsers().filter(user => `${user.name} ${user.email}` .toLowerCase() .includes(search) ); document.getElementById("usersList").innerHTML = users.length ? users.map(userHTML).join("") : "

No users found.

"; } function userHTML(user) { return `

${escapeHTML(user.name)}

Email: ${escapeHTML(user.email)}

Status: ${escapeHTML(user.status || "active")}

Wallet: ${money(user.wallet)}

User ID: ${escapeHTML(user.id)}

`; } // ----------------------------- // USER DETAILS // ----------------------------- function viewUser(id) { const user = getUsers().find(u => u.id === id); if (!user) { alert("User not found."); return; } const transactions = getTransactions().filter( t => t.userId === id ); app.innerHTML = `

User Details

${escapeHTML(user.name)}

Email: ${escapeHTML(user.email)}

Status: ${escapeHTML(user.status)}

Wallet: ${money(user.wallet)}

Joined: ${formatDate(user.joined)}

User Transactions

${ transactions.length ? transactions.map(transactionHTML).join("") : "

No transactions for this user.

" }
`; } // ----------------------------- // USER STATUS // ----------------------------- function toggleUserStatus(id) { const users = getUsers(); const user = users.find(u => u.id === id); if (!user) return; user.status = user.status === "active" ? "inactive" : "active"; saveUsers(users); showUsers(); } // ----------------------------- // WALLET FUNDING // ----------------------------- function showWalletFunding(preselectedUser = "") { const users = getUsers(); app.innerHTML = `

Fund User Wallet

`; } function fundWallet() { const userId = document.getElementById("fundUser").value; const amount = Number(document.getElementById("fundAmount").value); if (!userId) { alert("Please select a user."); return; } if (!amount || amount <= 0) { alert("Enter a valid amount."); return; } const users = getUsers(); const user = users.find(u => u.id === userId); if (!user) { alert("User not found."); return; } user.wallet = Number(user.wallet || 0) + amount; saveUsers(users); const transactions = getTransactions(); transactions.push({ id: createID("TXN"), userId: user.id, userName: user.name, type: "Wallet Funding", amount, status: "completed", date: new Date().toISOString(), description: "Admin wallet funding" }); saveTransactions(transactions); alert( `${money(amount)} added to ${user.name}'s wallet.` ); showUsers(); } // ----------------------------- // TRANSACTIONS // ----------------------------- function showTransactions() { const transactions = getTransactions(); app.innerHTML = `

Transaction History

${ transactions.length ? transactions .slice() .reverse() .map(transactionHTML) .join("") : "

No transactions yet.

" }
`; } function filterTransactions() { const search = document .getElementById("transactionSearch") .value .toLowerCase(); const type = document.getElementById("transactionType").value; const transactions = getTransactions().filter(t => { const matchesSearch = `${t.id} ${t.userName} ${t.type} ${t.status}` .toLowerCase() .includes(search); const matchesType = !type || t.type === type; return matchesSearch && matchesType; }); document.getElementById("transactionsList").innerHTML = transactions.length ? transactions .slice() .reverse() .map(transactionHTML) .join("") : "

No matching transactions.

"; } function transactionHTML(t) { return `

${escapeHTML(t.type)}

Transaction ID: ${escapeHTML(t.id)}

User: ${escapeHTML(t.userName || "Unknown")}

Amount: ${money(t.amount)}

Status: ${escapeHTML(t.status)}

Date: ${formatDate(t.date)}

${ t.description ? `

${escapeHTML(t.description)}

` : "" }
`; } // ----------------------------- // ADD TRANSACTION // ----------------------------- function showAddTransaction() { const users = getUsers(); app.innerHTML = `

Add Transaction

`; } function addTransaction() { const userId = document.getElementById("transactionUser").value; const type = document.getElementById("transactionService").value; const amount = Number( document.getElementById("transactionAmount").value ); const status = document.getElementById("transactionStatus").value; const description = document.getElementById( "transactionDescription" ).value; if (!userId) { alert("Select a user."); return; } if (amount < 0) { alert("Enter a valid amount."); return; } const user = getUsers().find(u => u.id === userId); if (!user) { alert("User not found."); return; } const transactions = getTransactions(); transactions.push({ id: createID("TXN"), userId, userName: user.name, type, amount, status, description, date: new Date().toISOString() }); saveTransactions(transactions); alert("Transaction saved."); showTransactions(); } // ----------------------------- // START DASHBOARD // ----------------------------- createDemoData(); renderDashboard();
