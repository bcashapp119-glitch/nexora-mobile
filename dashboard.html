async function getSupportResponse(question) { try { const response = await fetch( "https://nexora-ai-backend-omega.vercel.app/api/ai/support", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ message: question }) } ); const data = await response.json(); if (data.success) { return data.reply; } return "Sorry, Nexora AI is temporarily unavailable."; } catch (error) { console.error(error); return "Sorry, I couldn't connect to Nexora AI right now."; } }
